  function project3D(x, y, z, w, h) {

    // Rotate around Y

    const cosY = Math.cos(rotY), sinY = Math.sin(rotY);

    const x1 = x * cosY - z * sinY;

    const z1 = x * sinY + z * cosY;

    // Rotate around X

    const cosX = Math.cos(rotX), sinX = Math.sin(rotX);

    const y2 = y * cosX - z1 * sinX;

    const z2 = y * sinX + z1 * cosX;

    const fov = 2.6;

    const scale = (Math.min(w, h) * 0.46 * zoom) * (fov / (fov + z2 + 1.2));

    return {

      sx: w * 0.5 + x1 * scale,

      sy: h * 0.49 - y2 * scale,

      z: z2,

      scale

    };

  }

      chakras.forEach((c, idx) => {

        const pulse = 0.055 + Math.sin(t * 2 + idx) * 0.008;

        ctx.strokeStyle = c.col;

        ctx.lineWidth = 2;

        ctx.beginPath();

        for (let i = 0; i <= 20; i++) {

          const a = (i / 20) * Math.PI * 2;

          const pr = project3D(Math.cos(a) * pulse, c.y, Math.sin(a) * pulse, w, h);

          if (i === 0) ctx.moveTo(pr.sx, pr.sy); else ctx.lineTo(pr.sx, pr.sy);

        }

        ctx.stroke();

      });

    }

    // 6. Draw Reiki Byosen Scanning Rings

    if (layers.reiki) {

      for (let r = 0; r < 2; r++) {

        const scanY = Math.sin(t * 1.2 + r * 2) * 0.75;

        ctx.strokeStyle = "rgba(52, 211, 153, 0.75)";

        ctx.lineWidth = 2.2;

        ctx.beginPath();

        for (let i = 0; i <= 28; i++) {

          const a = (i / 28) * Math.PI * 2;

          const pr = project3D(Math.cos(a) * 0.28, scanY, Math.sin(a) * 0.22, w, h);

          if (i === 0) ctx.moveTo(pr.sx, pr.sy); else ctx.lineTo(pr.sx, pr.sy);

        }

        ctx.stroke();

      }

    }

    // 7. Draw Toroidal Warding Shield

    if (layers.shield) {

      ctx.strokeStyle = "rgba(244, 63, 94, 0.45)";

      ctx.lineWidth = 1.4;

      for (let m = 0; m < 6; m++) {

        const meridAng = (m / 6) * Math.PI + t * 0.4;

        ctx.beginPath();

        for (let i = 0; i <= 28; i++) {

          const a = (i / 28) * Math.PI * 2;

          const rad = Math.sin(a) * 0.52;

          const px = Math.cos(meridAng) * rad;

          const pz = Math.sin(meridAng) * rad;

          const py = Math.cos(a) * 1.02;

          const pr = project3D(px, py, pz, w, h);

          if (i === 0) ctx.moveTo(pr.sx, pr.sy); else ctx.lineTo(pr.sx, pr.sy);

        }

        ctx.stroke();

      }

    }

    // 8. Draw Clickable 3D Acupoints, Chakras & Auricular Nodes

    projectedHitBoxes = [];

    NODES.forEach(n => {

      if (n.cat === "chakra" && !layers.chakras) return;

      if (n.cat !== "chakra" && !layers.points) return;

      const coords = [n.pos];

      if (n.mirror) coords.push([-n.pos[0], n.pos[1], n.pos[2]]);

      coords.forEach((pt, sideIdx) => {

        const pr = project3D(pt[0], pt[1], pt[2], w, h);

        const isSel = n.id === selectedId;

        const isDiag = highlightedIds.has(n.id);

        let fill = n.cat === "chakra" ? "#c084fc" : (n.cat === "aur" ? "#34d399" : "#f59e0b");

        let radius = isSel ? 7.5 : (isDiag ? 7.0 : 4.8);

        if (isSel) fill = "#38bdf8";

        else if (isDiag) fill = "#10b981";

        // Outer glow

        if (isSel || isDiag) {

          ctx.fillStyle = isSel ? "rgba(56, 189, 248, 0.32)" : "rgba(16, 185, 129, 0.32)";

          ctx.beginPath();

          ctx.arc(pr.sx, pr.sy, radius + 5, 0, Math.PI * 2);

          ctx.fill();

        }

        ctx.fillStyle = fill;

        ctx.strokeStyle = "#020617";

        ctx.lineWidth = 1.5;

        ctx.beginPath();

        ctx.arc(pr.sx, pr.sy, radius, 0, Math.PI * 2);

        ctx.fill();

        ctx.stroke();

        // Callout label on primary side

        if ((isSel || isDiag) && sideIdx === 0) {

          ctx.fillStyle = "#f8fafc";

          ctx.font = "bold 11px sans-serif";

          ctx.fillText(n.code.split(" ")[0], pr.sx + 9, pr.sy + 4);

        }

        projectedHitBoxes.push({ id: n.id, x: pr.sx, y: pr.sy, r: 13 });

      });

    });

    requestAnimationFrame(draw3DScene);

  }
